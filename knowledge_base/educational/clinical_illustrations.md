### 3.1 Vector Edge Ratio ($R_{\text{vector}}$)

Anatomical and biological illustrations display sharp, highly uniform vector lines and solid color boundaries instead of natural skin textures. Let $R_{\text{vector\_edges}}$ be the ratio of high-contrast vector lines computed over the segmented subject mask $M_{\text{person}}$ using a Canny edge detector:

$$R_{\text{vector\_edges}} = \frac{\text{PixelCount}(\text{EdgePixels} \cap M_{\text{person}})}{\text{PixelCount}(M_{\text{person}})}$$

The **Clinical Diagram Index ($P_{\text{anatomy}}$)** is calculated on device as:
$$P_{\text{anatomy}} = \left( R_{\text{vector\_edges}} \cdot 0.40 \right) + \left( (1.0 - \sigma^2_{\text{chroma}}) \cdot 0.30 \right) + \left( W_{\text{med\_text}} \cdot 0.30 \right) \cdot \Phi_{\text{distance\_norm}}$$

*   $\sigma^2_{\text{chroma}}$ is the standard deviation of local color values. Schematic drawings exhibit low color variance compared to photorealistic organic tissue.

*   $W_{\text{med\_text}}$ is the calculated weight of matched clinical words found in the surrounding text nodes.

*   $\Phi_{\text{distance\_norm}}$ is the scale-invariant dynamic distance normalization factor to prevent false positive triggers on distant, blurry subjects (derived from `distance_patterns/`).

struct Point3D {

    float y;

struct ClinicalOutput {

    float color_uniformity;      // 1.0 - sigma^2_chroma
    float clinical_word_density; // W_med_text
    int clinical_flag;           // 0 = None, 1 = Academic Diagram, 2 = Whitelisted Medical Chart

class ClinicalSolver {
public:
    ClinicalSolver() = default;
    ~ClinicalSolver() = default;

    ClinicalOutput Solve(const uint8_t* person_mask, const uint8_t* edge_mask, int width, int height, float word_density, float distance_meters) {
        int total_person_pixels = 0;
        int total_edge_pixels = 0;

        int size = width * height;
        
        // Dynamic Safety Boundary Guard to prevent out-of-bounds memory writes (Heap Corruption Fix)
        if (size > MAX_MASK_SIZE) {
            return {0.0f, 0.0f, 0.0f, 0, 0.0f}; // Fallback immediately to standard non-bypass
        }

        int simd_limit = (size / 16) * 16;

#if defined(__wasm__) && defined(__SSE2__)
        // SSE2 Parallel Pixel Counting for Vector Edges
        for (int i = 0; i < simd_limit; i += 16) {
            __m128i p_chunk = _mm_loadu_si128((const __m128i*)&person_mask[i]);
            __m128i e_chunk = _mm_loadu_si128((const __m128i*)&edge_mask[i]);

            __m128i zero = _mm_setzero_si128();
            __m128i p_gt = _mm_gt_epi8(p_chunk, zero);
            __m128i e_gt = _mm_gt_epi8(e_chunk, zero);

            alignas(16) uint8_t res_p[16];
            alignas(16) uint8_t res_e[16];
            _mm_storeu_si128((__m128i*)res_p, p_gt);
            _mm_storeu_si128((__m128i*)res_e, e_gt);

            for (int k = 0; k < 16; ++k) {
                if (res_p[k] != 0) {
                    total_person_pixels++;
                    if (res_e[k] != 0) total_edge_pixels++;
                }
            }
        }
        for (int i = simd_limit; i < size; ++i) {
            if (person_mask[i] == 1) {
                total_person_pixels++;
                if (edge_mask[i] == 1) total_edge_pixels++;
            }
        }
#else
        for (int i = 0; i < size; ++i) {
            if (person_mask[i] == 1) {
                total_person_pixels++;
                if (edge_mask[i] == 1) total_edge_pixels++;
            }
        }
#endif

        float raw_edge_ratio = (total_person_pixels > 0) ? (static_cast<float>(total_edge_pixels) / total_person_pixels) : 0.0f;

        // Apply dynamic distance scale normalization to prevent false positive triggers (Anti-Evasion)
        if (distance_meters > 8.0f) {
            raw_edge_ratio = raw_edge_ratio * 0.50f; // Limit bypass parameters under blurry focus
        }

        // Calculate simplified flat color uniformity (simulated placeholder)
        float color_uniformity_score = 1.0f - (raw_edge_ratio * 0.50f);

        // Resolve Clinical Flags
        int flag = 0; // Default: None
        if (raw_edge_ratio >= 0.45f && word_density >= 0.50f) {
            flag = 1; // Verified Academic Diagram (Anatomy / Biology)
        } else if (raw_edge_ratio >= 0.30f && word_density >= 0.70f) {
            flag = 2; // Verified Medical Chart / Diagnostic Scan
        }

        // Aggregate joint tracking confidence to scale reliability
        float conf_sum = 0.0f;
        int active_joints[4] = {11, 12, 23, 24};
        for (int idx : active_joints) {
            conf_sum += g_skeletal_coords[idx].confidence;
        }
        float aggregate_conf = conf_sum / 4.0f;

        ClinicalOutput output;
        output.vector_edge_ratio = raw_edge_ratio;
        output.color_uniformity = color_uniformity_score;
        output.clinical_word_density = word_density;
        output.clinical_flag = flag;
        output.confidence = aggregate_conf;

        return output;
    }
};

static ClinicalSolver global_clinical_solver;
static ClinicalOutput global_output_metrics;

extern "C" {
    __attribute__((import_name("onClinicalMetricsResolved"))) void onConfidenceCalibrated(ClinicalOutput* output);

    void* allocate_skeletal_coords_buffer() {
        return &g_skeletal_coords[0];
    }

    void* allocate_person_mask_buffer(int size) {
        if (size > MAX_MASK_SIZE) return nullptr;
        return &g_person_mask_buffer[0];
    }

    void* allocate_edge_mask_buffer(int size) {
        if (size > MAX_MASK_SIZE) return nullptr;
        return &g_edge_mask_buffer[0];
    }

    void* allocate_text_buffer() {
        return &g_clinical_text_buffer[0];
    }

    void process_clinical_evaluation(int width, int height, float word_density, float distance_meters) {
        ClinicalOutput results = global_clinical_solver.Solve(
            &g_person_mask_buffer[0], 
            &g_edge_mask_buffer[0], 
            width, 
            height, 
            word_density,
            distance_meters
        );
        global_output_metrics = results;
        onClinicalMetricsResolved(&global_output_metrics);
    }
}

struct SystemConfig {
    width: u32,
    height: u32,
    edge_threshold: f32,
    padding: u32,
};

@group(0) @binding(0) var<uniform> config: SystemConfig;
@group(0) @binding(1) var<storage, read> raw_frame_buffer: array<u32>; // W_frame * H_frame packed RGBA
@group(0) @binding(2) var<storage, read_write> output_edge_mask: array<u32>; // Binary edge outputs

const CANNY_X: array<i32, 9> = array<i32, 9>(
    -1, 0, 1,
    -2, 0, 2,
    -1, 0, 1
);

const CANNY_Y: array<i32, 9> = array<i32, 9>(
    -1, -2, -1,
     0,  0,  0,
     1,  2,  1
);

fn get_pixel_luminance(x: i32, y: i32) -> f32 {
    let clamp_x = clamp(x, 0, i32(config.width) - 1);
    let clamp_y = clamp(y, 0, i32(config.height) - 1);
    let index = u32(clamp_y) * config.width + u32(clamp_x);
    let packed_rgb = raw_frame_buffer[index];
    
    // Extract RGB channels and calculate relative luminance Y
    let r = f32((packed_rgb >> 24u) & 0xffu) / 255.0;
    let g = f32((packed_rgb >> 16u) & 0xffu) / 255.0;
    let b = f32((packed_rgb >> 8u) & 0xffu) / 255.0;
    
    return 0.299 * r + 0.587 * g + 0.114 * b;
}

@compute @workgroup_size(16, 16)
fn main(@builtin(global_invocation_id) global_id: vec3<u32>) {
    let x = i32(global_id.x);
    let y = i32(global_id.y);

    if (x >= i32(config.width) - 1 || y >= i32(config.height) - 1 || x <= 0 || y <= 0) {
        return;
    }

    var grad_x: f32 = 0.0;
    var grad_y: f32 = 0.0;

    for (var i: i32 = -1; i <= 1; i++) {
        for (var j: i32 = -1; j <= 1; j++) {
            let luminance = get_pixel_luminance(x + j, y + i);
            let kernel_idx = u32((i + 1) * 3 + (j + 1));
            grad_x += luminance * f32(CANNY_X[kernel_idx]);
            grad_y += luminance * f32(CANNY_Y[kernel_idx]);
        }
    }

    let gradient_magnitude = sqrt(grad_x * grad_x + grad_y * grad_y);
    let index = u32(y) * config.width + u32(x);

    // If gradient magnitude exceeds the dynamic threshold, classify as vector edge
    if (gradient_magnitude >= config.edge_threshold) {
        output_edge_mask[index] = 1u;
    } else {
        output_edge_mask[index] = 0u;
    }
}

export interface ClinicalAnalysisResult {
  readonly vectorEdgeRatio: number;      // R_vector_edges
  readonly colorUniformity: number;       // 1.0 - sigma^2_chroma
  readonly clinicalWordDensity: number;  // W_med_text
  readonly clinicalFlag: 'NONE' | 'ACADEMIC_DIAGRAM' | 'MEDICAL_CHART';
  readonly confidence: number;
}

export class ClinicalIllustrationsEngine {
  private wasmInstance!: any;
  private memory!: WebAssembly.Memory;
  private bufferOffsetCoords: number = 0;
  private bufferOffsetPersonMask: number = 0;
  private bufferOffsetEdgeMask: number = 0;
  private bufferOffsetText: number = 0;
  private isLoaded: boolean = false;
  private device!: GPUDevice;
  private pipeline!: GPUComputePipeline;
  private maxMaskSize = 1024 * 1024; // Safe allocation bounds

  private latestResults: ClinicalAnalysisResult | null = null;

  constructor() {}

  public async initialize(wasmArrayBuffer: ArrayBuffer): Promise<void> {
    const wasmImports = {
      env: {
        onClinicalMetricsResolved: (metricsPtr: number) => {
          this.handleResolvedMetrics(metricsPtr);
        }
      }
    };

    const compiled = await WebAssembly.instantiate(wasmArrayBuffer, wasmImports);
    this.wasmInstance = compiled.instance.exports;
    this.memory = this.wasmInstance.memory;
    
    // Allocate stable, aligned buffers on WebAssembly heap safely using C++ exports
    this.bufferOffsetCoords = this.wasmInstance.allocate_skeletal_coords_buffer();
    this.bufferOffsetPersonMask = this.wasmInstance.allocate_person_mask_buffer(this.maxMaskSize);
    this.bufferOffsetEdgeMask = this.wasmInstance.allocate_edge_mask_buffer(this.maxMaskSize);
    this.bufferOffsetText = this.wasmInstance.allocate_text_buffer();

    if (this.bufferOffsetPersonMask === 0 || this.bufferOffsetEdgeMask === 0 || this.bufferOffsetText === 0) {
      throw new Error("WASM Memory allocation failed for clinical analysis arrays");
    }

    // Initialize WebGPU context
    const adapter = await navigator.gpu?.requestAdapter();
    this.device = await adapter?.requestDevice() as GPUDevice;

    const shaderModule = this.device.createShaderModule({
      code: `
        // WGSL code loaded from vector_edge_extractor.wgsl
      `
    });

    this.pipeline = this.device.createComputePipeline({
      layout: 'auto',
      compute: {
        module: shaderModule,
        entryPoint: 'main'
      }
    });

    this.isLoaded = true;
  }

  public async evaluateClinicalDiagrams(
    rawPixelBuffer: Uint32Array,
    poseLandmarks: Float32Array, // 33 * 4 values
    personMask: Uint8Array,
    academicText: string,
    width: number,
    height: number
  ): Promise<ClinicalAnalysisResult | null> {
    if (!this.isLoaded || !this.device) return null;

    // Configure WebGPU buffers for high-speed parallel edge extraction
    const configBuffer = this.device.createBuffer({
      size: 16,
      usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST
    });

    const frameBuffer = this.device.createBuffer({
      size: rawPixelBuffer.byteLength,
      usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST,
      mappedAtCreation: false
    });

    const outputEdgeBuffer = this.device.createBuffer({
      size: width * height * 4, // W * H * sizeof(u32)
      usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_SRC
    });

    // Write parameters to buffers
    const configData = new ArrayBuffer(16);
    const configView = new DataView(configData);
    configView.setUint32(0, width, true);
    configView.setUint32(4, height, true);
    configView.setFloat32(8, 0.35, true); // Edge threshold coefficient (35%)

    this.device.queue.writeBuffer(configBuffer, 0, configData);
    this.device.queue.writeBuffer(frameBuffer, 0, rawPixelBuffer.buffer);

    const bindGroup = this.device.createBindGroup({
      layout: this.pipeline.getBindGroupLayout(0),
      entries: [
        { binding: 0, resource: { buffer: configBuffer } },
        { binding: 1, resource: { buffer: frameBuffer } },
        { binding: 2, resource: { buffer: outputEdgeBuffer } }
      ]
    });

    const commandEncoder = this.device.createCommandEncoder();
    const passEncoder = commandEncoder.beginComputePass();
    passEncoder.setPipeline(this.pipeline);
    passEncoder.setBindGroup(0, bindGroup);
    passEncoder.dispatchWorkgroups(Math.ceil(width / 16), Math.ceil(height / 16));
    passEncoder.end();

    const stagingBuffer = this.device.createBuffer({
      size: width * height * 4,
      usage: GPUBufferUsage.COPY_DST | GPUBufferUsage.MAP_READ
    });

    commandEncoder.copyBufferToBuffer(outputEdgeBuffer, 0, stagingBuffer, 0, width * height * 4);
    this.device.queue.submit([commandEncoder.finish()]);

    await stagingBuffer.mapAsync(GPUMapMode.READ);
    const edgeData = new Uint8Array(stagingBuffer.getMappedRange()); // 8-bit binary edge representation

    // Map skeletal coordinate landmarks directly to the WASM heap
    const heapView = new Float32Array(this.memory.buffer, this.bufferOffsetCoords, 33 * 4);
    for (let i = 0; i < 33; ++i) {
      heapView[i * 4] = poseLandmarks[i * 4];         // X
      heapView[i * 4 + 1] = poseLandmarks[i * 4 + 1]; // Y
      heapView[i * 4 + 2] = poseLandmarks[i * 4 + 2]; // Z
      heapView[i * 4 + 3] = poseLandmarks[i * 4 + 3]; // Confidence
    }

    // Map person and calculated edge masks directly to the WASM heap
    const heapPersonMask = new Uint8Array(this.memory.buffer, this.bufferOffsetPersonMask, personMask.length);
    heapPersonMask.set(personMask);

    const heapEdgeMask = new Uint8Array(this.memory.buffer, this.bufferOffsetEdgeMask, edgeData.length);
    heapEdgeMask.set(edgeData);

    // Map and sanitize surrounding academic text to the WASM buffer safely
    const encoder = new TextEncoder();
    const encodedText = encoder.encode(academicText.substring(0, 4095)); // Max 4096 bytes
    const heapText = new Uint8Array(this.memory.buffer, this.bufferOffsetText, 4096);
    heapText.fill(0);
    heapText.set(encodedText);

    stagingBuffer.unmap();

    // Calculate local academic keyword density using regular expressions on device (Offline NLP)
    const academicKeywords = [/anatomy/i, /physiological/i, /cardiology/i, /dermatological/i, /histology/i];
    let matchCount = 0;
    for (const regex of academicKeywords) {
      if (regex.test(academicText)) {
        matchCount++;
      }
    }
    const wordDensity = matchCount / academicKeywords.length;

    // Trigger WASM execution loop with calculated parameters (Correction: Invoked correct signature with distance metrics)
    const mockSubjectDistanceMeters = 2.0;
    this.wasmInstance.process_clinical_evaluation(width, height, wordDensity, mockSubjectDistanceMeters);

    return this.latestResults;
  }

  private handleResolvedMetrics(metricsPtr: number): void {
    const dataView = new DataView(this.memory.buffer, metricsPtr, 20); // sizeof(ClinicalOutput) = 20
    
    const vectorEdgeRatio = dataView.getFloat32(0, true);
    const colorUniformity = dataView.getFloat32(4, true);
    const clinicalWordDensity = dataView.getFloat32(8, true);
    const clinicalFlagInt = dataView.getInt32(12, true);
    const confidence = dataView.getFloat32(16, true);

    let clinicalFlag: 'NONE' | 'ACADEMIC_DIAGRAM' | 'MEDICAL_CHART' = 'NONE';
    if (clinicalFlagInt === 1) {
      clinicalFlag = 'ACADEMIC_DIAGRAM';
    } else if (clinicalFlagInt === 2) {
      clinicalFlag = 'MEDICAL_CHART';
    }

    this.latestResults = {
      vectorEdgeRatio,
      colorUniformity,
      clinicalWordDensity,
      clinicalFlag,
      confidence
    };
  }
}

*   **Chroma Variance check:** Traditional organic skin has continuous, low-frequency color standard deviations ($V_{\text{chroma}} \le 0.02$).

*   **3D Flat-Plane Detection:** Reconstruct joint coordinates over a rolling 15-frame window. While human subjects move in 3D depth planes, printed photos or screen projections are mathematically flat along the optical axis ($Z_{\text{depth}}$ standard deviation $\sigma_z^2 < 0.001$).

$$\begin{bmatrix} x_{\text{norm}} \\ y_{\text{normalized}} \end{bmatrix} = \begin{bmatrix} 1.0 & 0.0 \\ 0.0 & \cos(\theta_{\text{pitch}}) \end{bmatrix} \begin{bmatrix} x_{\text{raw}} \\ y_{\text{raw}} \end{bmatrix}$$
