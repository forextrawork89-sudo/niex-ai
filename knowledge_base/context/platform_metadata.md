# Platform Metadata Intelligence Framework

This document is the authoritative metadata reasoning specification for the entire context subsystem.
It replaces prior metadata discussions with a research-grade Platform Metadata Intelligence Framework that explains what metadata is, how it contributes to context and intent understanding, how it contributes to confidence and uncertainty estimation, and why metadata must be treated as evidence rather than truth.
The framework preserves conceptual reasoning and metadata evidence modeling while excluding implementation-specific runtime, deployment, and hardware optimization discussion.

## 2. Platform Taxonomy

### 2.0 Social-media safety evidence patterns
The canonical KB should treat short-video and social-media content as a high-risk environment for contextual manipulation. The following patterns deserve additional scrutiny even when they do not contain explicit sexual or violent keywords:

- Suggestive bait: captions that promise exclusive access, secret content, or private reveals
- Pressure tactics: urgency, countdowns, limited-time rewards, or implied exclusivity
- Gambling persuasion: easy-money or guaranteed-profit claims with strong call-to-action
- Scam/phishing framing: account verification, password reset, bonus claims, fake prizes
- Euphemistic sexual promotion: indirect invitation language that signals sexualized access or private content

These signals should be considered weak-to-moderate evidence unless reinforced by stronger signals such as explicit intent, sexualized visual context, or coercive language.

```cpp
constexpr int METADATA_BUFFER_SIZE = 4096;

struct MetadataInput {
    float domain_risk_rating;     // R_domain
    float title_lexical_risk;     // R_title
    float opengraph_tags_risk;    // R_og
    int active_domain_flag;       // 0 = Neutral, 1 = Whitelisted Academic, 2 = Blocked Domain
};

struct MetadataOutput {
    float compiled_metadata_risk; // S_meta
    int resolved_safety_state;    // 0 = TRUSTED, 1 = NEUTRAL, 2 = UNTRUSTED
    float confidence;
};

// Global memory buffers mapped to WebAssembly Linear Memory Heap
MetadataInput g_metadata_input;
MetadataOutput g_metadata_output;
char g_scraped_header_buffer[METADATA_BUFFER_SIZE];

class PlatformMetadataSolver {
public:
    PlatformMetadataSolver() = default;
    ~PlatformMetadataSolver() = default;

    MetadataOutput Solve(const MetadataInput& input, const char* header_text, int text_length) {
        float r_domain = std::clamp(input.domain_risk_rating, 0.0f, 1.0f);
        float r_title = std::clamp(input.title_lexical_risk, 0.0f, 1.0f);
        float r_og = std::clamp(input.opengraph_tags_risk, 0.0f, 1.0f);

        float w_dom = 0.50f;
        float w_title = 0.25f;
        float w_og = 0.25f;

        // Calculate compiled metadata risk score (S_meta)
        float s_meta = (r_domain * w_dom) + (r_title * w_title) + (r_og * w_og);
        s_meta = std::clamp(s_meta, 0.0f, 1.0f);

        // Parse local scraped headers to detect educational/academic markers (Linguistic boost)
        float academic_boost = 0.0f;
        if (text_length > 0) {
            // Simple offline prefix-tree search simulation for critical keywords
            std::string text(header_text, text_length);
            if (text.find("journal") != std::string::npos || text.find("university") != std::string::npos || text.find("science") != std::string::npos) {
                academic_boost = 0.30f;
            }
        }

        s_meta = s_meta - academic_boost;
        s_meta = std::clamp(s_meta, 0.0f, 1.0f);

        int state = 1; // NEUTRAL
        if (input.active_domain_flag == 1 || s_meta < 0.30f) {
            state = 0; // TRUSTED (Relax thresholds)
        } else if (input.active_domain_flag == 2 || s_meta >= 0.75f) {
            state = 2; // UNTRUSTED (Escalate checking rates)
        }

        MetadataOutput output;
        output.compiled_metadata_risk = s_meta;
        output.resolved_safety_state = state;
        output.confidence = 1.0f - (std::abs(r_domain - r_title) * 0.20f);

        return output;
    }
};

static PlatformMetadataSolver global_metadata_solver;

extern "C" {
    __attribute__((import_name("onMetadataMetricsResolved"))) void onConfidenceCalibrated(MetadataOutput* output);

    }
```

```typescript
export interface MetadataAnalysisResult {
  readonly compiledMetadataRisk: number; // S_meta
  readonly resolvedSafetyState: 'TRUSTED' | 'NEUTRAL' | 'UNTRUSTED';
  readonly confidence: number;
}

export class PlatformMetadataEngine {
  private wasmInstance!: any;
  private memory!: WebAssembly.Memory;
  private bufferOffsetInput = 0;
  private bufferOffsetText = 0;
  private isLoaded = false;
  private latestResults: MetadataAnalysisResult | null = null;

  constructor() {}

  public async initialize(wasmArrayBuffer: ArrayBuffer): Promise<void> {
    const wasmImports = {
      env: {
        onMetadataMetricsResolved: (metricsPtr: number) => {
          this.handleResolvedMetrics(metricsPtr);
        }
      }
    };

    const compiled = await WebAssembly.instantiate(wasmArrayBuffer, wasmImports);
    this.wasmInstance = compiled.instance.exports;
    this.memory = this.wasmInstance.memory;

    // Allocate buffers on WebAssembly heap
    this.bufferOffsetInput = this.wasmInstance.allocate_metadata_input_buffer();
    this.bufferOffsetText = this.wasmInstance.allocate_header_text_buffer();

    if (this.bufferOffsetInput === 0 || this.bufferOffsetText === 0) {
      throw new Error("WASM Memory allocation failed for metadata analysis buffers");
    }

    this.isLoaded = true;
  }

  public async evaluateMetadataContext(
    domainRiskRating: number,
    titleLexicalRisk: number,
    opengraphTagsRisk: number,
    activeDomainFlag: number,
    scrapedHeaderText: string
  ): Promise<MetadataAnalysisResult | null> {
    if (!this.isLoaded) return null;

    // Map input fields to WASM heap (MetadataInput struct)
    const viewInput = new DataView(this.memory.buffer, this.bufferOffsetInput, 16);
    viewInput.setFloat32(0, domainRiskRating, true);
    viewInput.setFloat32(4, titleLexicalRisk, true);
    viewInput.setFloat32(8, opengraphTagsRisk, true);
    viewInput.setInt32(12, activeDomainFlag, true);

    // Map header text to WASM buffer safely
    const encoder = new TextEncoder();
    const encodedText = encoder.encode(scrapedHeaderText.substring(0, 4095)); // Max 4096 bytes
    const heapText = new Uint8Array(this.memory.buffer, this.bufferOffsetText, 4096);
    heapText.fill(0);
    heapText.set(encodedText);

    // Execute WASM evaluation
    this.wasmInstance.process_metadata_evaluation(encodedText.length);

    return this.latestResults;
  }

  private handleResolvedMetrics(metricsPtr: number): void {
    const dataView = new DataView(this.memory.buffer, metricsPtr, 12); // sizeof(MetadataOutput) = 12
    
    const compiledMetadataRisk = dataView.getFloat32(0, true);
    const resolvedSafetyStateInt = dataView.getInt32(4, true);
    const confidence = dataView.getFloat32(8, true);

    let resolvedSafetyState: 'TRUSTED' | 'NEUTRAL' | 'UNTRUSTED' = 'NEUTRAL';
    if (resolvedSafetyStateInt === 0) {
      resolvedSafetyState = 'TRUSTED';
    } else if (resolvedSafetyStateInt === 2) {
      resolvedSafetyState = 'UNTRUSTED';
    }

    this.latestResults = {
      compiledMetadataRisk,
      resolvedSafetyState,
      confidence
    };
  }
}
```
