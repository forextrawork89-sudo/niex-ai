# Suggestive Combination Intelligence Framework
This document is the authoritative research knowledge base for understanding suggestive combinations in apparel perception.
It is not an implementation guide. It is a foundational conceptual framework for explainable clothing reasoning.
The central thesis is that suggestive combinations emerge from garment geometry, textile behavior, human body interaction, environmental context, motion, and temporal stability rather than from any single cue.

## 1. Purpose, Scope, and Foundational Intent
This framework exists to explain how clothing combinations become visually suggestive, semantically charged, context-dependent, or benign.
It defines recognition logic, semantic interpretation, uncertainty handling, temporal reasoning, and explainability for suggestive combinations.
Its purpose is to preserve a research-grade ontology that supports future multimodal clothing understanding systems.

### 1.1 Operational Definition
A suggestive combination is a garment arrangement whose meaning is shaped by interaction between coverage, fabric transparency, tightness, posture, movement, context, and viewer interpretation.
It is not determined by a single exposed area percentage, single detector score, or isolated body-region cue.
A combination becomes suggestive when multiple streams of evidence reinforce a stronger semantic reading than ordinary clothing would support.

### 1.2 Non-Scope
This document does not describe runtime optimization, browser execution, JavaScript orchestration, WebAssembly inference, hardware acceleration, or detector engineering.
This document preserves conceptual knowledge needed for explainable clothing intelligence.

A garment pairing is not merely the sum of two visual objects. It produces a new semantic structure when body geometry, layer order, fabric behavior, and context are considered together.

// Double Exponential Smoothing Filter for the output risk scores
class TemporalDoubleSmoother {
public:
    TemporalDoubleSmoother() : is_initialized_(false), alpha_(0.70f), beta_(0.20f) {}

    float Apply(float raw_value) {
        if (!is_initialized_) {
            s_ = raw_value;
            b_ = 0.0f;
            is_initialized_ = true;
            return raw_value;
        }

        float prev_s = s_;
        s_ = alpha_ * raw_value + (1.0f - alpha_) * (s_ + b_);
        b_ = beta_ * (s_ - prev_s) + (1.0f - beta_) * b_;
        return s_;
    }

    void Reset() {
        is_initialized_ = false;
    }

private:
    bool is_initialized_;
    float alpha_;
    float beta_;
    float s_;
    float b_;
};

class ApparelBeliefNetwork {
public:
    ApparelBeliefNetwork() {
        smoother_.Reset();
    }
    ~ApparelBeliefNetwork() = default;

    JointRiskOutput Evaluate(const InputApparelIndices& ind) {
        // Clamp and normalize input indicators
        float i_exposure = std::clamp(ind.exposure_index, 0.0f, 1.0f);
        float i_compression = std::clamp(ind.compression_index, 0.0f, 1.0f);
        float i_transparency = std::clamp(ind.transparency_index, 0.0f, 1.0f);
        float m_context = std::clamp(ind.context_multiplier, 0.40f, 1.40f);

        // Define conditional weights
        float w_exp = 0.45f;
        float w_comp = 0.35f;
        float w_trans = 0.20f;

        // Apply Bayesian Joint Probability Multiplication
        float term_exp = 1.0f - w_exp * i_exposure;
        float term_comp = 1.0f - w_comp * i_compression;
        float term_trans = 1.0f - w_trans * i_transparency;

        float raw_joint_risk = 1.0f - (term_exp * term_comp * term_trans);

        // Apply environmental context scaling
        raw_joint_risk = raw_joint_risk * m_context;
        raw_joint_risk = std::clamp(raw_joint_risk, 0.0f, 1.0f);

        // Apply temporal double exponential smoothing
        float smoothed_risk = smoother_.Apply(raw_joint_risk);

        // Resolve system override actions
        int override_level = 0; // PASS
        if (smoothed_risk >= 0.70f) {
            override_level = 2; // BLOCK_BLUR
        } else if (smoothed_risk >= 0.45f) {
            override_level = 1; // ESCALATE_SENSITIVITY
        }

        JointRiskOutput out;
        out.joint_risk_score = raw_joint_risk;
        out.smoothed_risk_score = smoothed_risk;
        out.system_override_level = override_level;
        out.confidence = 1.0f - (std::abs(i_exposure - i_compression) * 0.25f); // High discrepancies slightly degrade certainty

        return out;
    }

    void Reset() {
        smoother_.Reset();
    }

private:
    TemporalDoubleSmoother smoother_;
};

export interface JointRiskAnalysisResult {
  readonly jointRiskScore: number;       // R_combo
  readonly smoothedRiskScore: number;
  readonly confidence: number;
  readonly systemAction: 'PASS' | 'ESCALATE_SENSITIVITY' | 'BLOCK_BLUR';
}

export class SuggestiveCombinationsEngine {
  private wasmInstance!: any;
  private memory!: WebAssembly.Memory;
  private bufferOffsetIndices: number = 0;
  private bufferOffsetPersonMask: number = 0;
  private bufferOffsetApparelMask: number = 0;
  private isLoaded: boolean = false;
  private device!: GPUDevice;
  private pipeline!: GPUComputePipeline;
  private patchDimension = 128;
  private maxMaskSize = 1024 * 1024; // Safe allocation bounds

  private latestResults: JointRiskAnalysisResult | null = null;

  constructor() {}

  public async initialize(wasmArrayBuffer: ArrayBuffer): Promise<void> {
    const wasmImports = {
      env: {
        onJointRiskMetricsResolved: (metricsPtr: number) => {
          this.handleResolvedMetrics(metricsPtr);
        }
      }
    };

    const compiled = await WebAssembly.instantiate(wasmArrayBuffer, wasmImports);
    this.wasmInstance = compiled.instance.exports;
    this.memory = this.wasmInstance.memory;
    
    // Allocate stable, aligned buffers on WebAssembly heap safely using C++ exports
    this.bufferOffsetIndices = this.wasmInstance.allocate_apparel_indices_buffer();
    this.bufferOffsetPersonMask = this.wasmInstance.allocate_person_mask_buffer(this.maxMaskSize);
    this.bufferOffsetApparelMask = this.wasmInstance.allocate_apparel_mask_buffer(this.maxMaskSize);

    if (this.bufferOffsetPersonMask === 0 || this.bufferOffsetApparelMask === 0) {
      throw new Error("WASM Memory allocation failed for masks array");
    }

    // Initialize WebGPU context
    const adapter = await navigator.gpu?.requestAdapter();
    this.device = await adapter?.requestDevice() as GPUDevice;

    const shaderModule = this.device.createShaderModule({
      code: `
        // WGSL code loaded from cooccurrence_feature_reducer.wgsl
      `
    });

    this.pipeline = this.device.createComputePipeline({
      layout: 'auto',
      compute: {
        module: shaderModule,
        entryPoint: 'main'
      }
    });

    this.isLoaded = true;
  }

  public async evaluateJointRisks(
    personMask: Uint8Array,
    apparelMask: Uint8Array,
    exposureIndex: number,     // I_1
    compressionIndex: number,  // I_2
    transparencyIndex: number, // I_3
    contextMultiplier: number, // M_context
    width: number,
    height: number
  ): Promise<JointRiskAnalysisResult | null> {
    if (!this.isLoaded || !this.device) return null;

    // Configure WebGPU buffers for high-speed parallel feature reduction
    const configBuffer = this.device.createBuffer({
      size: 16,
      usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST
    });

    const pMaskBuffer = this.device.createBuffer({
      size: personMask.byteLength,
      usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST,
      mappedAtCreation: false
    });

    const aMaskBuffer = this.device.createBuffer({
      size: apparelMask.byteLength,
      usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST,
      mappedAtCreation: false
    });

    const outputFeatureBuffer = this.device.createBuffer({
      size: this.patchDimension * this.patchDimension * 4, // 128 * 128 * sizeof(f32)
      usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_SRC
    });

    // Write parameters to buffers
    const configData = new Uint32Array([width, height, this.patchDimension, 0]);
    this.device.queue.writeBuffer(configBuffer, 0, configData.buffer);
    this.device.queue.writeBuffer(pMaskBuffer, 0, personMask.buffer);
    this.device.queue.writeBuffer(aMaskBuffer, 0, apparelMask.buffer);

    const bindGroup = this.device.createBindGroup({
      layout: this.pipeline.getBindGroupLayout(0),
      entries: [
        { binding: 0, resource: { buffer: configBuffer } },
        { binding: 1, resource: { buffer: pMaskBuffer } },
        { binding: 2, resource: { buffer: aMaskBuffer } },
        { binding: 3, resource: { buffer: outputFeatureBuffer } }
      ]
    });

    const commandEncoder = this.device.createCommandEncoder();
    const passEncoder = commandEncoder.beginComputePass();
    passEncoder.setPipeline(this.pipeline);
    passEncoder.setBindGroup(0, bindGroup);
    passEncoder.dispatchWorkgroups(Math.ceil(width / 16), Math.ceil(height / 16));
    passEncoder.end();

    const stagingBuffer = this.device.createBuffer({
      size: this.patchDimension * this.patchDimension * 4,
      usage: GPUBufferUsage.COPY_DST | GPUBufferUsage.MAP_READ
    });

    commandEncoder.copyBufferToBuffer(outputFeatureBuffer, 0, stagingBuffer, 0, this.patchDimension * this.patchDimension * 4);
    this.device.queue.submit([commandEncoder.finish()]);

    await stagingBuffer.mapAsync(GPUMapMode.READ);
    stagingBuffer.unmap();

    // Map raw metrics directly to the WASM heap
    const heapView = new Float32Array(this.memory.buffer, this.bufferOffsetIndices, 4); // 4 float fields
    heapView[0] = exposureIndex;
    heapView[1] = compressionIndex;
    heapView[2] = transparencyIndex;
    heapView[3] = contextMultiplier;

    // Map person and apparel masks directly to the WASM heap
    const heapPersonMask = new Uint8Array(this.memory.buffer, this.bufferOffsetPersonMask, personMask.length);
    heapPersonMask.set(personMask);

    const heapApparelMask = new Uint8Array(this.memory.buffer, this.bufferOffsetApparelMask, apparelMask.length);
    heapApparelMask.set(apparelMask);

    // Trigger WASM execution loop (Correction: Corrected to call the exact solver export)
    this.wasmInstance.process_joint_risk_evaluation();

    return this.latestResults;
  }

  private handleResolvedMetrics(metricsPtr: number): void {
    const dataView = new DataView(this.memory.buffer, metricsPtr, 16); // sizeof(JointRiskOutput) = 16
    
    const jointRiskScore = dataView.getFloat32(0, true);
    const smoothedRiskScore = dataView.getFloat32(4, true);
    const confidence = dataView.getFloat32(8, true);
    const systemOverrideLevelInt = dataView.getInt32(12, true);

    let systemAction: 'PASS' | 'ESCALATE_SENSITIVITY' | 'BLOCK_BLUR' = 'PASS';
    if (systemOverrideLevelInt === 2) {
      systemAction = 'BLOCK_BLUR';
    } else if (systemOverrideLevelInt === 1) {
      systemAction = 'ESCALATE_SENSITIVITY';
    }

    this.latestResults = {
      jointRiskScore,
      smoothedRiskScore,
      confidence,
      systemAction
    };
  }
}
